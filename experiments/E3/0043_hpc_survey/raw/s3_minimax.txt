#!/usr/bin/env bash

set -a
# shellcheck disable=SC1090
source "$PROFILE"
set +a

node="$(hostname -s)"
case "$node" in
  hpcgpu06) comfy_port=18606; api_port=8406 ;;
  hpcgpu07) comfy_port=18607; api_port=8407 ;;
  *) echo "Unsupported node for this launcher: $node" >&2; exit 64 ;;
esac

instance="a100-1gpu-${node}"
dit_path="$BASE/ComfyUI/models/diffusion_models/$MINIMAX_H3_DIT_NAME"
bf16_path="$BASE/ComfyUI/models/diffusion_models/minimax_h3_ref2va_pruned_bf16.safetensors"
token_path="$BASE/run/api-token-$node"

mkdir -p "$BASE"/{logs,run,outputs} \
  "$BASE/instances/$instance"/{user,input,output,temp,torchinductor}

echo "[$(date -u +%FT%TZ)] MiniMax H3 FP16 start attempt on $node" >&2
echo "profile: precision=$MINIMAX_H3_PRECISION resolution=$MINIMAX_H3_DEFAULT_RESOLUTION steps=$MINIMAX_H3_STEPS easycache=$MINIMAX_H3_ENABLE_CACHE attention=$MINIMAX_H3_ATTENTION_BACKEND concurrency=$MINIMAX_H3_API_CONCURRENCY" >&2
nvidia-smi --query-gpu=index,name,memory.total,memory.free --format=csv,noheader >&2

[[ -x "$ENV_DIR/bin/python" && -x "$ENV_DIR/bin/uvicorn" ]] || {
  echo "MiniMax H3 environment is incomplete: $ENV_DIR" >&2
  exit 65
}
if [[ ! -s "$dit_path" ]]; then
  if [[ ! -s "$bf16_path" ]]; then
    echo "Official BF16 source checkpoint is not ready: $bf16_path" >&2
    echo "The allocation keeper will retain the GPU and retry without falling back to FP8." >&2
    exit 66
  fi
  echo "FP16 checkpoint is absent; acquiring the shared conversion lock." >&2
  exec 9>"$BASE/run/minimax-h3-fp16-conversion.lock"
  flock 9
  if [[ ! -s "$dit_path" ]]; then
    "$ENV_DIR/bin/python" "$BASE/convert_minimax_h3_bf16_to_fp16.py" "$bf16_path" "$dit_path"
  fi
  flock -u 9
fi

token="$($ENV_DIR/bin/python -c 'import secrets; print(secrets.token_hex(32))')"
printf '%s\n' "$token" >"$token_path"

export CUDA_VISIBLE_DEVICES=0 CUDA_DEVICE_ORDER=PCI_BUS_ID
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True
export TORCHINDUCTOR_CACHE_DIR="$BASE/instances/$instance/torchinductor"
export MINIMAX_H3_DIT_COMPUTE_DEVICE=cuda:0
export MINIMAX_H3_DIT_DONOR_DEVICE=cpu
export MINIMAX_H3_CLIP_DEVICE=cpu
export MINIMAX_H3_VAE_DEVICE=cuda:0
export MINIMAX_H3_API_TOKEN="$token"
export MINIMAX_H3_UPSTREAM_URL="http://127.0.0.1:$comfy_port"

comfy_pid=""
api_pid=""
cleanup_children() {
  [[ -n "$api_pid" ]] && kill "$api_pid" 2>/dev/null || true
  [[ -n "$comfy_pid" ]] && kill "$comfy_pid" 2>/dev/null || true
  wait 2>/dev/null || true
}
trap cleanup_children EXIT INT TERM

wait_for_service() {
  local url="$1" pid="$2" name="$3" deadline=$((SECONDS + 1800))
  until curl --connect-timeout 2 -fsS "$url" >/dev/null 2>&1; do
    kill -0 "$pid" 2>/dev/null || { echo "$name exited before healthy" >&2; return 1; }
    (( SECONDS < deadline )) || { echo "Timed out waiting for $name" >&2; return 1; }
    sleep 3
  done
}

cd "$BASE/ComfyUI"
"$ENV_DIR/bin/python" main.py \
  --listen 127.0.0.1 --port "$comfy_port" --disable-auto-launch \
  --preview-method none --disable-metadata --fp16-unet --use-sage-attention \
  --user-directory "$BASE/instances/$instance/user" \
  --input-directory "$BASE/instances/$instance/input" \
  --output-directory "$BASE/instances/$instance/output" \
  --temp-directory "$BASE/instances/$instance/temp" \
  --database-url "sqlite:///$BASE/instances/$instance/user/comfyui.db" \
  >"$BASE/logs/comfy-$instance.log" 2>&1 &
comfy_pid=$!
wait_for_service "http://127.0.0.1:$comfy_port/system_stats" "$comfy_pid" "ComfyUI-$node"

cd "$BASE"
"$ENV_DIR/bin/uvicorn" api:create_app --factory \
  --host 0.0.0.0 --port "$api_port" \
  >"$BASE/logs/api-$instance.log" 2>&1 &
api_pid=$!
wait_for_service "http://127.0.0.1:$api_port/health" "$api_pid" "MiniMax-H3-API-$node"

echo "[$(date -u +%FT%TZ)] MiniMax H3 ready on $node: API port $api_port, default 1366x768 / 30 steps" >&2
wait "$api_pid"

=================== ~/h3_guard.sh
#!/usr/bin/env bash
# H3 6卡任务自我守护：只要它不存在(R/PD 都没有)，就先清理 park 占位、再立即重提。
# 目的：把 H3 崩了/结束导致的"卡释放窗口"压缩到 1 分钟内，防止别人趁隙抢占。
BASE="${H3_BASE:-/home/mail/chenyuma/minimax-h3-fp8}"
SB="$BASE/minimax-h3-6g.sbatch"
LOG="$BASE/logs/h3_guard.log"

# 只要还有(任意状态) minimax-h3-6g 任务，就不动它
ANY=$(squeue -h -u chenyuma -n minimax-h3-6g 2>/dev/null | wc -l)
[ "$ANY" -ge 1 ] && exit 0

# 清掉会占位的 park 空占任务(gpu_park_4 / gpu_park_a100)，给 H3 腾 6 张
PARK=$(squeue -h -u chenyuma -t RUNNING -o "%i %j" 2>/dev/null | awk  /Users/chenyuma /^gpu_park/ {print })
[ -n "$PARK" ] && { scancel $PARK 2>/dev/null; echo "$(date +%F %T) 已取消park占位: $PARK" >> "$LOG"; }

# 重提 H3 6卡任务
sleep 2
if sbatch "$SB" >> "$LOG" 2>&1; then
  echo "$(date +%F %T) H3 6卡任务已自动复活" >> "$LOG"
else
  echo "$(date +%F %T) H3 复活失败" >> "$LOG"
fi

#!/usr/bin/env bash
# 定时任务守护：保证「minimax-h3-a100-2gpu」作业始终处于 SUSP 队列（占用 2 张 A100）。
#
# 由 crontab 每 5 分钟调用一次；逻辑：
#   1. 若该作业已在 R(运行)/PD(排队) => 不做任何事。
#   2. 若不存在或已终止 => 重新提交 sbatch，令其回到排队，一旦 A100 释放立即抢占。
#
# crontab 行示例（登录节点，minute 级）：
#   */5 * * * * /home/mail/chenyuma/minimax-h3-fp8/schedule_a100_h3.sh >>/home/mail/chenyuma/minimax-h3-fp8/logs/schedule-a100.log 2>&1
set -euo pipefail

BASE="${MINIMAX_H3_BASE:-/home/mail/chenyuma/minimax-h3-fp8}"
JOB_NAME="minimax-h3-a100-2gpu"
SBATCH="$BASE/minimax-h3-a100-2gpu.sbatch"
LOG_DIR="$BASE/logs"
mkdir -p "$LOG_DIR"

# 已存在（运行中或排队中）则不重复提交
if squeue --name="$JOB_NAME" --states=RUNNING,PENDING -h | grep -q .; then
  echo "$(date -u +%FT%TZ) 作业已在队列/运行中，跳过。"
  exit 0
fi

echo "$(date -u +%FT%TZ) 未发现作业，重新提交 $SBATCH"
cd "$BASE" && sbatch "$SBATCH"
